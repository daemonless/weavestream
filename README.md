## Deployment

### Podman Compose

**1.** Save as `.env` and fill in what is empty:

```env { data-zip-bundle="weavestream-podman" data-zip-filename=".env" }
# Weavestream environment variables
# Copy to .env and adjust as needed

POSTGRES_USER=weavestream
POSTGRES_PASSWORD=weavestream
POSTGRES_DB=weavestream

DATA_DIR=./data
WEB_HOST_PORT=3000
APP_URL=http://localhost:3000
API_URL=http://localhost:3000/api

PUID=1000
PGID=1000
TZ=UTC
```

**2.** Save as `compose.yaml`:

```yaml { data-zip-bundle="weavestream-podman" data-zip-filename="compose.yaml" }
name: weavestream

services:
  postgres:
    image: ghcr.io/daemonless/postgres:16
    container_name: weavestream-postgres
    restart: always
    network_mode: host
    annotations:
      org.freebsd.jail.allow.sysvipc: "true"
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-weavestream}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-weavestream}
      POSTGRES_DB: ${POSTGRES_DB:-weavestream}
      PUID: ${PUID:-1000}
      PGID: ${PGID:-1000}
      TZ: ${TZ:-UTC}
    volumes:
      - ${DATA_DIR:-./data}/postgres:/var/db/postgres/data16

  redis:
    image: ghcr.io/daemonless/redis:latest
    container_name: weavestream-redis
    restart: always
    network_mode: host
    environment:
      PUID: ${PUID:-1000}
      PGID: ${PGID:-1000}
      TZ: ${TZ:-UTC}
    volumes:
      - ${DATA_DIR:-./data}/redis:/data

  api:
    image: ghcr.io/daemonless/weavestream-api:latest
    container_name: weavestream-api
    restart: always
    network_mode: host
    depends_on:
      - postgres
      - redis
    environment:
      PUID: ${PUID:-1000}
      PGID: ${PGID:-1000}
      TZ: ${TZ:-UTC}
      DATABASE_URL: postgresql://${POSTGRES_USER:-weavestream}:${POSTGRES_PASSWORD:-weavestream}@127.0.0.1:5432/${POSTGRES_DB:-weavestream}
      REDIS_URL: redis://127.0.0.1:6379/0
      APP_URL: ${APP_URL:-http://localhost:3000}
      API_URL: ${API_URL:-http://localhost:3000/api}
      FILE_STORAGE_DIR: /var/lib/weavestream/files
      BACKUP_STORAGE_DIR: /var/lib/weavestream/backup
      JWT_SIGNING_KEY: ${JWT_SIGNING_KEY:-}
      MFA_ENCRYPTION_KEY: ${MFA_ENCRYPTION_KEY:-}
      PASSWORD_ENCRYPTION_KEY: ${PASSWORD_ENCRYPTION_KEY:-}
      INTEGRATION_SECRET_KEY: ${INTEGRATION_SECRET_KEY:-}

      COOKIE_SIGNING_KEY: ${COOKIE_SIGNING_KEY:-}
      CSRF_SIGNING_KEY: ${CSRF_SIGNING_KEY:-}

      - ${DATA_DIR:-./data}/files:/var/lib/weavestream/files

  worker:
    image: ghcr.io/daemonless/weavestream-worker:latest
    container_name: weavestream-worker
    restart: always
    network_mode: host
    depends_on:
      - redis

      TZ: ${TZ:-UTC}

      REDIS_URL: redis://127.0.0.1:6379/0
      APP_URL: ${APP_URL:-http://localhost:3000}
      API_URL: ${API_URL:-http://localhost:3000/api}
      FILE_STORAGE_DIR: /var/lib/weavestream/files

      SMTP_SECRET_KEY: ${SMTP_SECRET_KEY:-}
      COOKIE_SIGNING_KEY: ${COOKIE_SIGNING_KEY:-}
      CSRF_SIGNING_KEY: ${CSRF_SIGNING_KEY:-}
    volumes:
      - ${DATA_DIR:-./data}/files:/var/lib/weavestream/files
      - ${DATA_DIR:-./data}/backup:/var/lib/weavestream/backup

      PUID: ${PUID:-1000}
      PGID: ${PGID:-1000}
      TZ: ${TZ:-UTC}
      API_INTERNAL_URL: http://127.0.0.1:4000
```

| `PGID` | `1000` | Group ID for the container processes |

| `API_URL` | `http://localhost:3000/api` | Public URL of the API (default: http://localhost:3000/api) |
| `DATA_DIR` | `./data` | Host path for persistent data (files, backups, database, redis) |
| `WEB_HOST_PORT` | `3000` | Published host port for the web UI (default: 3000) |
